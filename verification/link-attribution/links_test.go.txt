func TestFormalInlineBound(t *testing.T) {
	for opener := 1; opener <= 4; opener++ {
		for closer := 1; closer <= 4; closer++ {
			text := strings.Repeat("`", opener) + "[x](#x)" + strings.Repeat("`", closer)
			want := 1
			if opener == closer {
				want = 0
			}
			if got := len(ExtractLinks(text)); got != want {
				t.Fatalf("%q got %d want %d", text, got, want)
			}
		}
	}
	if len(ExtractLinks("[a `code`](#a)")) != 1 || len(InlineCodeSpans("`a``b``c``")) != 1 || len(InlineCodeSpans("`unclosed")) != 0 {
		t.Fatal("inline pairing or contained code changed")
	}
}

func TestFormalCodeCursor(t *testing.T) {
	for a := 0; a < 6; a++ {
		for b := a + 1; b <= 6; b++ {
			for c := b; c < 7; c++ {
				for d := c + 1; d <= 7; d++ {
					spans := [][2]int{{a, b}, {c, d}}
					for firstStart := 0; firstStart < 8; firstStart++ {
						for firstEnd := firstStart + 1; firstEnd <= 8; firstEnd++ {
							for start := firstStart; start < 8; start++ {
								for end := max(start+1, firstEnd); end <= 8; end++ {
									cursor := codeSpanCursor{spans: spans}
									cursor.overlaps(firstStart, firstEnd)
									if got, want := cursor.overlaps(start, end), overlapsCodeSpan(start, end, spans); got != want {
										t.Fatalf("spans=%v ranges=%d:%d,%d:%d got %v want %v", spans, firstStart, firstEnd, start, end, got, want)
									}
								}
							}
						}
					}
				}
			}
		}
	}
}

func TestFormalWrapScanner(t *testing.T) {
	fragments := []string{"[x](#x)", "[x](#)", "[x](#a[b](#c)", "[a [b](#x)", "`[x](#x)`", "[a `b`](#x)", "[x](#a`b)", "[x](#x)[y](#y)", "[x](#a)[z](#b)", "[x](#a", "[x](#a)[", "`", "``", " ", "z"}
	for _, left := range fragments {
		for _, right := range fragments {
			text := left + right
			scanner := newWrapLinkScanner(text)
			spans := InlineCodeSpans(text)
			for i := 0; i < len(text); i++ {
				want := 0
				loc := linkRegex.FindStringIndex(text[i:])
				if loc != nil && loc[0] == 0 && !overlapsCodeSpan(i, i+loc[1], spans) {
					want = loc[1]
				}
				if got := scanner.matchAt(i); got != want {
					t.Fatalf("scanner %q at %d got %d want %d", text, i, got, want)
				}
			}
		}
	}
}

func TestFormalPlacementBound(t *testing.T) {
	for _, prefix := range []string{"", "prose\n\n", "> ", "| "} {
		for _, label := range []string{"x", "long label", "界é", "a `code`"} {
			for count := 1; count <= 3; count++ {
				for width := 1; width <= 24; width++ {
					for height := 1; height <= 3; height++ {
						markup := "[" + label + "](#x)"
						source := prefix + strings.Repeat(markup+" ", count)
						b, err := Read(strings.NewReader(source), "x", width, height, false)
						if err != nil {
							t.Fatal(err)
						}
						for _, w := range []int{width, 25, width} {
							b.Reflow(w, height)
							links := 0
							labels := ""
							openers := 0
							for pi := 0; pi < b.PageCount(); pi++ {
								p := b.Page(pi)
								for _, link := range p.Links {
									if link.Target != "x" || link.Label != label || link.LineOnPage < 0 || link.LineOnPage >= len(p.Lines) {
										t.Fatalf("invalid link %+v", link)
									}
									links++
								}
								for _, span := range p.LabelSpans {
									if span.Line < 0 || span.Line >= len(p.Lines) || span.Start < 0 || span.End <= span.Start || span.End > len(p.Lines[span.Line]) || span.LinkIndex < -1 || span.LinkIndex >= len(p.Links) {
										t.Fatalf("invalid span %+v page %+v", span, p)
									}
									text := p.Lines[span.Line][span.Start:span.End]
									if !utf8.ValidString(text) {
										t.Fatalf("split rune %+v", span)
									}
									if span.Opener {
										openers++
										if text != "[" || span.LinkIndex < 0 {
											t.Fatalf("invalid opener %q %+v", text, span)
										}
									} else {
										labels += text
									}
								}
								for li, link := range p.Links {
									found := false
									for _, span := range p.LabelSpans {
										if span.LinkIndex == li && span.Line >= link.LineOnPage {
											found = true
										}
									}
									if !found {
										t.Fatalf("owner has no selectable label or opener: width=%d height=%d page=%+v", w, height, p)
									}
								}
							}
							if links != count || labels != strings.Repeat(label, count) {
								t.Fatalf("source=%q width=%d height=%d links=%d labels=%q want %q openers=%d", source, w, height, links, labels, strings.Repeat(label, count), openers)
							}
						}
					}
				}
			}
		}
	}
}

func TestFormalFenceBound(t *testing.T) {
	for _, source := range []string{"```\n[x](#x)\n```", "~~~\n[x](#x)\n~~~", "    [x](#x)", "```\n[x](#x)"} {
		b, err := Read(strings.NewReader(source), "x", 2, 1, false)
		if err != nil {
			t.Fatal(err)
		}
		for pi := 0; pi < b.PageCount(); pi++ {
			if len(b.Page(pi).Links) > 0 || len(b.Page(pi).LabelSpans) > 0 {
				t.Fatalf("code became selectable: %q", source)
			}
		}
	}
	b, err := Read(strings.NewReader("[x](#x)"), "x", 2, 1, true)
	if err != nil {
		t.Fatal(err)
	}
	for pi := 0; pi < b.PageCount(); pi++ {
		if len(b.Page(pi).Links) > 0 {
			t.Fatal("plain text became selectable")
		}
	}
}

func TestFormalAnchorBound(t *testing.T) {
	alphabet := []rune{'A', 'b', '1', ' ', '-', '!', '界'}
	var visit func(string, int)
	visit = func(s string, left int) {
		normalized := NormalizeAnchor(s)
		if NormalizeAnchor(normalized) != normalized || strings.HasPrefix(normalized, "-") || strings.HasSuffix(normalized, "-") || strings.Contains(normalized, "--") {
			t.Fatalf("invalid normalization %q -> %q", s, normalized)
		}
		for _, r := range normalized {
			if !(unicode.IsLower(r) || unicode.IsLetter(r) || unicode.IsDigit(r) || r == '-') {
				t.Fatalf("bad rune %q", r)
			}
		}
		if left > 0 {
			for _, r := range alphabet {
				visit(s+string(r), left-1)
			}
		}
	}
	visit("", 4)
	for input, want := range map[string]string{" A ! b- ": "a-b", "A!B": "ab", "界 界": "界-界"} {
		if got := NormalizeAnchor(input); got != want {
			t.Fatal(fmt.Sprintf("%q got %q want %q", input, got, want))
		}
	}
}

func TestFormalOccurrenceOrder(t *testing.T) {
	for width := 1; width <= 24; width++ {
		for height := 1; height <= 3; height++ {
			for _, source := range []string{"[first](#one) [second](#two) [third](#three)", "[first](#one)\n[second](#two)\n\n[third](#three)", "| [first](#one) [second](#two) [third](#three)"} {
				b, err := Read(strings.NewReader(source), "x", width, height, false)
				if err != nil {
					t.Fatal(err)
				}
				var targets []string
				for pi := 0; pi < b.PageCount(); pi++ {
					for _, link := range b.Page(pi).Links {
						targets = append(targets, link.Target)
					}
				}
				if strings.Join(targets, ",") != "one,two,three" {
					t.Fatalf("order source=%q width=%d height=%d got %v", source, width, height, targets)
				}
			}
		}
	}
}
