func compact(s string) string {
	return strings.Map(func(r rune) rune {
		if unicode.IsSpace(r) {
			return -1
		}
		return r
	}, s)
}

func formattedText(lines []string, width int) []string {
	doc := formatDocument(lines, width, false)
	result := make([]string, len(doc.lines))
	for i, line := range doc.lines {
		result[i] = line.text
	}
	return result
}

func TestGeneratedWrapConservation(t *testing.T) {
	alphabet := []string{"a", "b", "界", " ", "\t"}
	var visit func(string, int)
	visit = func(s string, n int) {
		for width := 1; width <= 5; width++ {
			got := WrapLines([]string{s}, width)
			if compact(strings.Join(got, "")) != compact(s) {
				t.Fatalf("lost text %q width%d: %q", s, width, got)
			}
			for _, line := range got {
				if runewidth.StringWidth(line) > max(width, 2) {
					t.Fatalf("overflow %q width%d", line, width)
				}
			}
		}
		if n > 0 {
			for _, c := range alphabet {
				visit(s+c, n-1)
			}
		}
	}
	visit("", 4)
}

func TestBlockContext(t *testing.T) {
	got := formattedText([]string{"# Heading", "", "Later prose"}, 40)
	if !reflect.DeepEqual(got, []string{"# Heading", "", "  Later prose"}) {
		t.Fatalf("first-block privilege reset: %q", got)
	}
	for _, fence := range []string{"```", "~~~~"} {
		b, err := Read(strings.NewReader(fence+"\n# Hidden\n\n"+fence+"\n# Visible"), "t", 40, 4, false)
		if err != nil {
			t.Fatal(err)
		}
		if b.PageForAnchor("hidden") != -1 || b.PageForAnchor("visible") < 0 {
			t.Fatal("fence leaked heading")
		}
	}
	for _, lines := range [][]string{{"Title", "==="}, {"Title", "---"}, {"Two", "line title", "==="}} {
		doc := formatDocument(lines, 40, false)
		if len(doc.anchors) != 1 || !strings.HasPrefix(doc.lines[0].text, "#") {
			t.Fatalf("setext lost: %+v", doc)
		}
	}
}

func TestLexicalBoundaries(t *testing.T) {
	for _, sep := range []string{"\n", "\r", "\r\n"} {
		got := splitLines([]byte("a" + sep + "b" + sep))
		if !reflect.DeepEqual(got, []string{"a", "b", ""}) {
			t.Fatalf("line ending %q: %q", sep, got)
		}
	}
	for s, want := range map[string]bool{"    x": true, "\tx": true, " \tx": true, "   x": false, "x": false, "": false} {
		if IsIndentedCodeLine(s) != want {
			t.Fatalf("indent %q", s)
		}
	}
	for s, want := range map[string]bool{"1. x": true, "12) x": true, "1.x": false, ". x": false, "12": false} {
		if isOrderedListItem(s) != want {
			t.Fatalf("list %q", s)
		}
	}
	if titleCase("hello world") != "Hello World" {
		t.Fatal("title boundary")
	}
}

func TestLayoutReflowAndPositions(t *testing.T) {
	texts := []string{"", "# A\n\none two three four five\n\n# A\nlast", "Title\n===\n\nword word\n\n    code", "```\n# Hidden\n```\n# End", "a\r\nb\rc\n"}
	for _, plain := range []bool{false, true} {
		for _, text := range texts {
			b, err := Read(strings.NewReader(text), "t", 5, 2, plain)
			if err != nil {
				t.Fatal(err)
			}
			for _, width := range []int{1, 2, 5, 20} {
				for _, height := range []int{1, 3} {
					b.Reflow(width, height)
					fresh, err := Read(strings.NewReader(text), "t", width, height, plain)
					if err != nil {
						t.Fatal(err)
					}
					if !reflect.DeepEqual(b.pages, fresh.pages) || !reflect.DeepEqual(b.pagePositions, fresh.pagePositions) || !reflect.DeepEqual(b.anchors, fresh.anchors) {
						t.Fatal("stale layout after reflow")
					}
					for page := 0; page < b.PageCount(); page++ {
						pos := b.PositionForPage(page)
						resolved := b.PageForPosition(pos)
						if resolved < 0 || resolved >= b.PageCount() {
							t.Fatal("position outside layout")
						}
						copy := b.Page(page)
						if len(copy.Lines) > 0 {
							copy.Lines[0] = "mutated"
							if b.Page(page).Lines[0] == "mutated" {
								t.Fatal("page aliases book")
							}
						}
					}
					for raw := -1; raw <= len(b.rawLines)+1; raw++ {
						p := b.PageForRawLine(raw)
						if p < 0 || p >= b.PageCount() {
							t.Fatal("raw lookup outside layout")
						}
					}
				}
			}
		}
	}
	formatted := []formattedLine{{text: "a", raw: 0}, {text: "bc", raw: 0}, {text: "", raw: -1}, {text: "z", raw: 1}}
	if got := lineSources(formatted); got[1].Offset != 1 || got[3].Offset != 0 {
		t.Fatalf("offset continuation %+v", got)
	}
	if got := rawLinePages(formatted, 1); got[0] != 0 {
		t.Fatal("first occurrence overwritten")
	}
	b, _ := Read(strings.NewReader("# A\n\npadding\n\n# A"), "t", 40, 1, false)
	if b.anchors["a"] != 4 {
		t.Fatal("duplicate heading did not overwrite")
	}
	if b.PageForAnchor("a") != 4 {
		t.Fatal("heading anchor map mismatch")
	}
	authoritative := lineSources([]formattedLine{{text: "ab", raw: 0}, {text: "c", raw: 0, sourceOffset: 7, hasOffset: true}})
	if authoritative[1].Offset != 7 {
		t.Fatalf("explicit offset ignored: %+v", authoritative)
	}

}

func TestProvenanceAndAnchorSelection(t *testing.T) {
	formatted := formatDocument([]string{"one two", "three four five"}, 4, false).lines
	last := -1
	for _, line := range formatted {
		if line.raw >= 0 {
			if line.raw < last {
				t.Fatal("provenance moved backward")
			}
			last = line.raw
		}
	}
	f := []formattedLine{{text: "start", raw: 0}, {text: "# heading", raw: 1, heading: true}}
	if pagePositions(f, 2)[0].Raw != 0 {
		t.Fatal("first page skipped start")
	}
	f = append(f, formattedLine{text: "", raw: -1}, formattedLine{text: "", raw: -1})
	p := pagePositions(f, 2)
	if p[1] != p[0] {
		t.Fatalf("blank page carry %+v", p)
	}
	for _, text := range []string{"  界", "[a b](#x)", "([long label](#target))", "a\tb", "``literal``"} {
		for width := 1; width <= 6; width++ {
			for _, line := range formattedText([]string{text}, width) {
				if runewidth.StringWidth(line) > max(width, 2) {
					t.Fatalf("indent fallback overflow %q width%d", line, width)
				}
			}
		}
	}
}

func TestTokenAndLookupFolds(t *testing.T) {
	var tokens []wrapToken
	eachWrapToken("  ([long label](#x))word", func(token wrapToken) { tokens = append(tokens, token) })
	if len(tokens) != 2 || tokens[0].text != "([long label](#x)" || !tokens[0].link || !tokens[0].spaceBefore || tokens[1].text != ")word" || tokens[1].spaceBefore {
		t.Fatalf("token boundary %+v", tokens)
	}
	joined := "a\t b c  d"
	mapped := mapWrappedProvenance([]wrappedLine{{text: "a b c"}, {text: "d"}}, joined, []int{0, 5}, 10, false)
	if mapped[0].raw != 10 || mapped[0].endRaw != 11 || mapped[0].sourceOffset != 0 || mapped[1].raw != 11 || mapped[1].sourceOffset != 1 {
		t.Fatalf("normalized provenance: %+v", mapped)
	}
	cursor := provenanceCursor{source: "a  b 界", offsets: []int{0, 5}}
	for i, want := range []Position{{Raw: 0, Offset: 0}, {Raw: 0, Offset: 1}, {Raw: 1, Offset: 0}, {Raw: 1, Offset: 1}} {
		if got := cursor.nextContent(); got != want {
			t.Fatalf("content cursor step%d: %+v want %+v", i, got, want)
		}
	}
	b := Book{pages: make([]Page, 3), pageHeight: 1, rawLinePages: map[int]int{0: 0, 2: 2}, lineSources: []Position{{Raw: 0}, {Raw: 0, Offset: 2}, {Raw: 2}}}
	for _, test := range []struct {
		position Position
		want     int
	}{{Position{Raw: 0, Offset: 1}, 0}, {Position{Raw: 0, Offset: 2}, 1}, {Position{Raw: 1, Offset: 5}, 0}, {Position{Raw: 3, Offset: 2}, 2}} {
		if got := b.PageForPosition(test.position); got != test.want {
			t.Fatalf("position %+v got%d want%d", test.position, got, test.want)
		}
	}
	for raw, want := range map[int]int{-1: 0, 0: 0, 1: 0, 2: 2, 3: 2} {
		if got := b.PageForRawLine(raw); got != want {
			t.Fatalf("raw%d got%d want%d", raw, got, want)
		}
	}
}

func TestPreservedLineBlocksAndHeadingFlags(t *testing.T) {
	for _, lines := range [][]string{{"| a | b |", "| c | d |"}, {"> alpha", "> beta"}} {
		doc := formatDocument(lines, 80, false)
		if len(doc.lines) != 2 || doc.lines[0].text != lines[0] || doc.lines[1].text != lines[1] || doc.lines[1].raw != 1 {
			t.Fatalf("line block grouping: %+v", doc.lines)
		}
	}
	doc := formatDocument([]string{"    # literal", "", "# long heading"}, 4, false)
	headingCount := 0
	for _, line := range doc.lines {
		if line.heading {
			headingCount++
		}
		if line.raw == 0 && line.heading {
			t.Fatal("code classified heading")
		}
	}
	if headingCount < 2 {
		t.Fatal("heading continuation lost classification")
	}
	b, err := Read(strings.NewReader("# Heading\n\n[x](#heading)"), "t", 40, 4, false)
	if err != nil {
		t.Fatal(err)
	}
	page := b.Page(0)
	if len(page.Headings) == 0 || len(page.Links) == 0 || len(page.LabelSpans) == 0 {
		t.Fatal("missing page metadata")
	}
	page.Headings[0] = false
	page.Links[0].Target = "changed"
	page.LabelSpans[0].Start = 999
	fresh := b.Page(0)
	if !fresh.Headings[0] || fresh.Links[0].Target == "changed" || fresh.LabelSpans[0].Start == 999 {
		t.Fatal("page metadata aliases book")
	}
}

func TestGeneratedNormalizedSourcePositions(t *testing.T) {
	inputs := []string{"a  b", "c\td", "界 e", " f g "}
	for _, first := range inputs {
		for _, second := range inputs {
			for width := 1; width <= 8; width++ {
				raw := []string{first, second}
				var source []Position
				for ri, text := range raw {
					offset := 0
					for _, r := range text {
						if !unicode.IsSpace(r) {
							source = append(source, Position{Raw: ri, Offset: offset})
							offset++
						}
					}
				}
				doc := formatDocument(raw, width, false)
				rank := 0
				for _, line := range doc.lines {
					count := nonSpaceRunes(line.text)
					if count == 0 {
						continue
					}
					if rank+count > len(source) {
						t.Fatal("formatted source grew")
					}
					want := source[rank]
					if line.raw != want.Raw || line.sourceOffset != want.Offset || !line.hasOffset || line.endRaw != source[rank+count-1].Raw {
						t.Fatalf("normalized source%q width%d row%+v want%+v end%d", raw, width, line, want, source[rank+count-1].Raw)
					}
					rank += count
				}
				if rank != len(source) {
					t.Fatal("formatted source lost content")
				}
			}
		}
	}
}
